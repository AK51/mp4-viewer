20class TemplateMatchingEngine: 

21 """ 

22 Manages template selection and matching for automatic object tracking. 

23  

24 The TemplateMatchingEngine is responsible for: 

25 - Storing and validating user-selected template images 

26 - Performing OpenCV template matching on video frames 

27 - Calculating match confidence scores 

28 - Tracking consecutive failed matches 

29 - Managing tracking state 

30  

31 Attributes: 

32 _template: The stored template image data 

33 _template_width: Width of the template in pixels 

34 _template_height: Height of the template in pixels 

35 _source_location: Original location where template was captured 

36 _last_match_location: Location of the most recent successful match 

37 _consecutive_failures: Count of consecutive frames with low confidence 

38 _tracking_active: Whether template tracking is currently active 

39 _confidence_threshold: Minimum confidence score for valid matches 

40 _max_consecutive_failures: Maximum failures before tracking loss

61 def set_template(self, template_image: Frame, source_location: Point) -> None: 

62 """ 

63 Store the user-selected template image and validate its size. 

64  

65 Per Requirement 5.1, captures the selected region as the template. 

66 Per Requirement 5.4, template sizes must be between 10x10 and 200x200 pixels. 

67 Per Requirement 5.5, templates smaller than 10x10 pixels are rejected. 

68 Per Requirement 5.3, new templates replace previously selected templates. 

69  

70 Args: 

71 template_image: Frame containing the template image data 

72 source_location: Point indicating where the template was captured 

73  

74 Raises: 

75 TemplateMatchingError: If template dimensions are invalid 

76 """ 

77 # Validate template dimensions 

78 width = template_image.width 

79 height = template_image.height 

80 

81 # Check minimum size (Requirement 5.5) 

82 if width < self.MIN_TEMPLATE_SIZE or height < self.MIN_TEMPLATE_SIZE: 

83 raise TemplateMatchingError( 

84 f"Template must be at least {self.MIN_TEMPLATE_SIZE}x{self.MIN_TEMPLATE_SIZE} pixels." 

85 ) 

86 

87 # Check maximum size (Requirement 5.4) 

88 if width > self.MAX_TEMPLATE_SIZE or height > self.MAX_TEMPLATE_SIZE: 

89 raise TemplateMatchingError( 

90 f"Template must not exceed {self.MAX_TEMPLATE_SIZE}x{self.MAX_TEMPLATE_SIZE} pixels." 

91 ) 

92 

93 # Store the template (Requirement 5.1, 5.3) 

94 self._template = template_image.data.copy() 

95 self._template_width = width 

96 self._template_height = height 

97 self._source_location = source_location 

98 

99 # Reset tracking state when new template is set 

100 self._last_match_location = source_location 

101 self._consecutive_failures = 0 

102 self._tracking_active = True

154 def match_template(self, frame: Frame) -> MatchResult: 

155 """ 

156 Perform template matching on the given frame using OpenCV. 

157  

158 Per Requirement 6.1, searches for the template in the frame. 

159 Per Requirement 6.4, uses OpenCV template matching with normalized cross-correlation. 

160 Per Requirement 6.5, confidence threshold of 0.6 for valid matches. 

161 Per Requirement 8.2, tracks consecutive failed matches. 

162  

163 Args: 

164 frame: The video frame to search for the template 

165  

166 Returns: 

167 MatchResult containing match status, location, confidence, and bounding box 

168  

169 Raises: 

170 TemplateMatchingError: If no template has been set 

171 """ 

172 # Check if template is set 

173 if self._template is None: 

174 raise TemplateMatchingError("No template selected. Please select a region to track.") 

175 

176 # Ensure frame is large enough to contain template 

177 if frame.width < self._template_width or frame.height < self._template_height: 

178 return MatchResult( 

179 found=False, 

180 location=None, 

181 confidence=0.0, 

182 bounding_box=None 

183 ) 

184 

185 # Perform template matching using normalized cross-correlation (Requirement 6.4) 

186 result = cv2.matchTemplate(frame.data, self._template, cv2.TM_CCOEFF_NORMED) 

187 

188 # Find the best match location 

189 min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result) 

190 

191 # max_val is the confidence score for TM_CCOEFF_NORMED 

192 confidence = float(max_val) 

193 

194 # Check if confidence meets threshold (Requirement 6.5) 

195 if confidence >= self.CONFIDENCE_THRESHOLD: 

196 # Match found - calculate center point 

197 top_left_x, top_left_y = max_loc 

198 center_x = top_left_x + self._template_width // 2 

199 center_y = top_left_y + self._template_height // 2 

200 

201 match_location = Point(x=center_x, y=center_y) 

202 bounding_box = Rectangle( 

203 x=top_left_x, 

204 y=top_left_y, 

205 width=self._template_width, 

206 height=self._template_height 

207 ) 

208 

209 # Update tracking state 

210 self._last_match_location = match_location 

211 self._consecutive_failures = 0 

212 

213 return MatchResult( 

214 found=True, 

215 location=match_location, 

216 confidence=confidence, 

217 bounding_box=bounding_box 

218 ) 

219 else: 

220 # Match confidence too low (Requirement 6.5) 

221 self._consecutive_failures += 1 

222 

223 # Check for tracking loss (Requirement 8.2) 

224 if self._consecutive_failures >= self.MAX_CONSECUTIVE_FAILURES: 

225 self._tracking_active = False 

226 

227 return MatchResult( 

228 found=False, 

229 location=self._last_match_location, # Maintain previous location 

230 confidence=confidence, 

231 bounding_box=None 

232 )
